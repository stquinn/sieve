// tail reports whether line closes the fenced region.
func (s *Scanner) tail(line string, col int) bool {
    if s.state != inFence {
        return false
    }
    trimmed := strings.TrimSpace(line)
    if col == 0 && !fence.MatchString(line) {
        return false // col-0 guard sees raw indentation
    }
    return fence.MatchString(trimmed)
}
